<Steps>
  <Step title="First Step">
    Navigate to the button on the top right where it Says "Login/Register".
    <Frame caption="Preview">
  ![](/images/account-creation/signup1.png)
</Frame>
  </Step>
  <Step title="Second Step">
    Click "Sign Up" and enter any username and password that is 8+ characters, lowercase, uppercase, number and special character.
    <Note>It doesn't have to be your vu username!</Note>
    <Frame caption="Preview">
    ![](/images/account-creation/signup2.png)
</Frame>
  </Step>
  <Step title="Third Step">
  Open your email provider to verify the account you entered in the previous step and press "Verify Email Address".
      <Frame caption="Preview">
    ![](/images/account-creation/signup3.png)
</Frame>
  </Step>
  <Step title="Fourth Step">
    You should see a prompt your account has been verified.
    <Frame caption="Preview">
  ![](/images/account-creation/signup4.png)
</Frame>
  </Step>
  <Step title="Fifth Step">
    Enter the email or username you provided in step 2 along with password.
    <Note>You can optionally check the box for "Remember Me" to save your login credentials, so you don't have to log in each time you visit VuArchives.com.</Note>
    <Frame caption="Preview">
  ![](/images/account-creation/signup5.png)
</Frame>
  </Step>
</Steps>

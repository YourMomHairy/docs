<Steps>
  <Step title="First Step">
    Navigate to the button on the top right, click your username and click "My Account". (Sign into your account if you haven't already)
    <Frame caption="Preview">
  ![](/images/account-deletion/delete1.png)
</Frame>
  </Step>
  <Step title="Second Step">
    Click the dropdown and you will see the option to "Delete Account"
    <Danger>Premium users must wait for their subscription to expire before deleting their account!</Danger>
    <Frame caption="Preview">
    ![](/images/account-deletion/delete2.png)
</Frame>
  </Step>
  <Step title="Third Step">
  You will get a message prompting you to enter your password, ensure to read the pop-up fully.
      <Frame caption="Preview">
    ![](/images/account-deletion/delete3.png)
</Frame>
  </Step>
</Steps>
